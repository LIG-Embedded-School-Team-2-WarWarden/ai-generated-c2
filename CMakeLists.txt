cmake_minimum_required(VERSION 3.24)
project(ai_generated_c2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(c2 INTERFACE)
target_include_directories(c2 INTERFACE include)
add_executable(c2_console src/main.cpp)
target_link_libraries(c2_console PRIVATE c2)
if(WIN32)
    add_executable(c2_sim_dashboard WIN32 src/win_main.cpp)
    target_link_libraries(c2_sim_dashboard PRIVATE c2 user32 gdi32)
endif()
enable_testing()
add_executable(c2_contract_tests tests/contracts.cpp)
target_link_libraries(c2_contract_tests PRIVATE c2)
add_test(NAME contracts COMMAND c2_contract_tests)

option(C2_BUILD_LIVE "Build the separate legacy v3 Windows runtime" OFF)
if(C2_BUILD_LIVE)
    if(NOT WIN32)
        message(FATAL_ERROR "The live transport currently requires Windows Winsock")
    endif()
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/external/warwarden-protocol/CMakeLists.txt")
        message(FATAL_ERROR "Run git submodule update --init --recursive first")
    endif()
    set(WARWARDEN_PROTOCOL_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    add_subdirectory(external/warwarden-protocol)
    file(GLOB C2_RUNTIME_SOURCES CONFIGURE_DEPENDS runtime/src/*.cpp)
    # Replace this source to integrate a team's solver without touching wire messages.
    set(C2_POINTING_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/runtime/src/pointing.cpp" CACHE FILEPATH "Pointing strategy implementation")
    list(REMOVE_ITEM C2_RUNTIME_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/runtime/src/pointing.cpp")
    add_library(c2_live STATIC ${C2_RUNTIME_SOURCES} ${C2_POINTING_SOURCE})
    target_include_directories(c2_live PUBLIC runtime/include)
    target_include_directories(c2_live PRIVATE vendor/lidar)
    target_compile_definitions(c2_live PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(c2_live PUBLIC WarWarden::Protocol ws2_32)
    add_executable(c2_server runtime/apps/server.cpp)
    target_link_libraries(c2_server PRIVATE c2_live)
    add_executable(dummy_observation runtime/apps/observation.cpp)
    target_link_libraries(dummy_observation PRIVATE c2_live)
    add_executable(dummy_effector runtime/apps/effector.cpp)
    target_link_libraries(dummy_effector PRIVATE c2_live)
    add_executable(c2_dashboard WIN32 src/live_dashboard.cpp)
    target_link_libraries(c2_dashboard PRIVATE c2_live user32 gdi32)
    target_include_directories(c2_dashboard PRIVATE src)
    add_executable(c2_manual_tests tests/manual_effector.cpp)
    target_link_libraries(c2_manual_tests PRIVATE c2_live)
    add_test(NAME manual_effector COMMAND c2_manual_tests)
    add_executable(c2_sequence_tests tests/asset_sequence.cpp)
    target_link_libraries(c2_sequence_tests PRIVATE c2_live)
    add_test(NAME asset_sequence COMMAND c2_sequence_tests)
    add_executable(c2_live_host_tests tests/live_host.cpp)
    target_link_libraries(c2_live_host_tests PRIVATE c2_live)
    target_include_directories(c2_live_host_tests PRIVATE src)
    add_test(NAME live_udp_host COMMAND c2_live_host_tests)
    set_tests_properties(live_udp_host PROPERTIES TIMEOUT 15)
endif()

option(C2_BUILD_DUMMY_TESTS "Enable optional Python equipment-double integration tests" ON)
if(C2_BUILD_DUMMY_TESTS)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/.venv-team/Scripts/python.exe")
        set(Python3_EXECUTABLE "${CMAKE_CURRENT_SOURCE_DIR}/.venv-team/Scripts/python.exe")
    elseif(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/.venv-team/bin/python")
        set(Python3_EXECUTABLE "${CMAKE_CURRENT_SOURCE_DIR}/.venv-team/bin/python")
    endif()
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_FOUND)
        add_test(NAME team_icd_mvp COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_native_mvp.py" --console $<TARGET_FILE:c2_mvp_console>)
        set_tests_properties(team_icd_mvp PROPERTIES TIMEOUT 120)
    endif()
endif()
